SELECT title FROM film
    WHERE title LIKE "M%"


SELECT title, description FROM film
    WHERE description LIKE "%Deleted Scenes%"


SELECT AVG(replacement_cost) FROM film
    WHERE description LIKE "%crocodile%"


SELECT title FROM film
    ORDER BY release_year
    LIMIT 10


SELECT rating, COUNT(*) FROM film
    GROUP BY rating


SELECT release_year FROM film
    GROUP BY release_year
    ORDER BY count(*) DESC


SELECT count(*) FROM film
    JOIN language ON film.language_id = language.language_id
    WHERE language.name = "French"


SELECT title, description FROM film
    JOIN film_category ON film_category.film_id = film.film_id
    JOIN category ON category.category_id = film_category.category_id
    WHERE category.name = "Comedy"
    ORDER BY length DESC
    LIMIT 1


SELECT count(*)/count_distinct(film.film_id)
    FROM film
    JOIN film_actor ON film.film_id = film_actor.film_id


SELECT DISTINCT title FROM film
    JOIN inventory ON film.film_id = inventory.film_id
    JOIN rental ON inventory.inventory_id = rental.inventory_id
    GROUP BY customer_id, film.film_id
    HAVING count(*) >= 3

SELECT DISTINCT customer.first_name, customer.last_name
    FROM customer, actor
    WHERE customer.first_name = actor.first_name AND customer.last_name = actor.last_name

SELECT DISTINCT address FROM address
    JOIN city ON city.city_id = address.city_id
    JOIN country ON country.country_id = city.country_id
    WHERE country = "Argentina"

SELECT MIN(film.release_year - actor.birth_year) AS age
    FROM film
    JOIN film_actor ON film_actor.film_id = film.film_id
    JOIN actor ON actor.actor_id = film_actor.actor_id


SELECT actor.last_name, actor.first_name
    FROM actor
    JOIN film_actor ON film_actor.actor_id = actor.actor_id
    JOIN film ON film.film_id = film_actor.film_id
    GROUP BY actor.actor_id
    ORDER BY count(*)
    LIMIT 5


SELECT title, avg(film.release_year - actor.birth_year) FROM film
    JOIN film_actor ON film_actor.film_id = film.film_id
    JOIN actor ON actor.actor_id = film_actor.actor_id
    GROUP BY film.film_id, film.release_year, actor.birth_year
    ORDER BY avg(film.release_year - actor.birth_year)
    LIMIT 10

SELECT DISTINCT last_name, first_name FROM actor
    JOIN film_actor ON film_actor.actor_id = actor.actor_id
    JOIN film ON film.film_id = film_actor.film_id
    JOIN film_category ON film_category.film_id = film.film_id
    JOIN category ON category.category_id = film_category.category_id
    WHERE name = "Horror"
